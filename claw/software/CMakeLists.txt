cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(claw_hand C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(claw_hand src/main.c src/feetech.c src/httpd.c src/gripper.c src/wire.c src/hcfg.c)
target_include_directories(claw_hand PRIVATE src ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(claw_hand pico_stdlib hardware_uart hardware_watchdog hardware_adc pico_cyw43_arch_lwip_threadsafe_background)
pico_enable_stdio_usb(claw_hand 1)
pico_add_extra_outputs(claw_hand)
