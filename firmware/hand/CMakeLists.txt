cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "Board type")
include(pico_sdk_import.cmake)
project(robokyle_hand C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(robokyle_hand src/main.c src/httpd.c src/feetech.c src/gripper.c src/wire.c src/hcfg.c src/json.c src/status_led.c)
pico_generate_pio_header(robokyle_hand ${CMAKE_CURRENT_LIST_DIR}/src/ws2812.pio)
target_include_directories(robokyle_hand PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/src)
target_compile_definitions(robokyle_hand PRIVATE CYW43_HOST_NAME="robokyle-hand" PICO_FLASH_ASSUME_CORE1_SAFE=1)
target_link_libraries(robokyle_hand pico_stdlib pico_flash hardware_adc hardware_uart hardware_pio hardware_watchdog
  pico_cyw43_arch_lwip_threadsafe_background)
pico_enable_stdio_usb(robokyle_hand 1)
pico_enable_stdio_uart(robokyle_hand 0)
pico_add_extra_outputs(robokyle_hand)
